    Create .github/workflows/build-image.yml. This is a NEW file.

    The workflow:
    - Triggers on workflow_dispatch (manual) and on push when SMOLBSD kernel config
      or smolfire-qemu.conf changes
    - Runs on <kvm-host>-amd64 self-hosted runner
    - Copies SMOLBSD configs from the checkout into the freebsd-src tree
    - Runs make vm-image with KERNCONF=SMOLBSD, WITHOUT_QEMU unset, NOPKG unset
    - Copies artifact to /home/studio/smolbsd-ci/smolbsd-amd64-tpm.qcow2
    - Writes SHA256 manifest

    ```yaml
    # SPDX-License-Identifier: Apache-2.0
    # build-image.yml — Build smolBSD amd64 TPM image on <kvm-host>.
    # Outputs: /home/studio/smolbsd-ci/smolbsd-amd64-tpm.qcow2
    #          /home/studio/smolbsd-ci/smolbsd-amd64-tpm.qcow2.sha256
    #
    # Trigger manually (workflow_dispatch) or automatically when SMOLBSD kernel
    # config or smolfire-qemu.conf changes.
    # After a successful build, run tpm-vm-test.yml to validate T1-T6.

    name: Build smolBSD TPM Image

    on:
      workflow_dispatch:
      push:
        paths:
          - sys/amd64/conf/SMOLBSD
          - release/tools/smolfire-qemu.conf

    env:
      FREEBSD_SRC:  /home/studio/bsd-build/src/freebsd-src
      WORLD_STAGE:  /home/studio/bsd-build/stage/amd64-postworld-20260411T130830
      OUTPUT_DIR:   /home/studio/smolbsd-ci
      OUTPUT_IMAGE: /home/studio/smolbsd-ci/smolbsd-amd64-tpm.qcow2

    jobs:
      build:
        name: Build smolBSD amd64 TPM qcow2
        runs-on: [self-hosted, linux, amd64, kvm]
        timeout-minutes: 120

        steps:
          - uses: actions/checkout@v4

          - name: Verify freebsd-src tree exists
            run: |
              test -d "$FREEBSD_SRC" || {
                echo "ERROR: freebsd-src not found at $FREEBSD_SRC"
                echo "Run initial setup on <kvm-host> to clone the FreeBSD source tree."
                exit 1
              }
              echo "freebsd-src found at $FREEBSD_SRC"

          - name: Copy SMOLBSD configs into freebsd-src
            run: |
              # KERNCONF=SMOLBSD requires sys/amd64/conf/SMOLBSD in the src tree
              cp -v sys/amd64/conf/SMOLBSD \
                "$FREEBSD_SRC/sys/amd64/conf/SMOLBSD"

              # CLOUDWARE_CONF=smolfire-qemu.conf requires it in release/tools/
              cp -v release/tools/smolfire-qemu.conf \
                "$FREEBSD_SRC/release/tools/smolfire-qemu.conf"

              # Verify copies
              grep "device tpm" "$FREEBSD_SRC/sys/amd64/conf/SMOLBSD"
              grep "VM_EXTRA_PACKAGES" "$FREEBSD_SRC/release/tools/smolfire-qemu.conf"
              echo "Configs copied successfully"

          - name: Create output directory
            run: mkdir -p "$OUTPUT_DIR"

          - name: Check for qemu-x86_64-static (needed for pkg chroot)
            run: |
              if [ -f /usr/bin/qemu-x86_64-static ]; then
                echo "QEMUSTATIC=/usr/bin/qemu-x86_64-static" >> "$GITHUB_ENV"
                echo "qemu-x86_64-static found — pkg chroot enabled"
              else
                echo "WARNING: /usr/bin/qemu-x86_64-static not found"
                echo "VM_EXTRA_PACKAGES install may fail without qemu-user-static"
                echo "To fix: sudo apt-get install -y qemu-user-static"
                echo "QEMUSTATIC=" >> "$GITHUB_ENV"
              fi

          - name: Build smolBSD amd64 TPM image
            run: |
              set -e
              cd "$FREEBSD_SRC"

              # CRITICAL: Do NOT set WITHOUT_QEMU, NOPKG, or NOPKGBASE.
              # These flags (used in run-freebsd-mini-pipeline.sh) skip pkg install
              # and would silently omit tpm2-tools from the image.
              # See plans/03-tpm-2-0-measured-boot/03-RESEARCH.md Pitfall 3.
              make -C release vm-image \
                KERNCONF=SMOLBSD \
                CLOUDWARE=smolbsd \
                CLOUDWARE_FLAGS="CLOUDWARE_CONF=smolfire-qemu.conf" \
                VMFORMATS=qcow2 \
                VMSIZE=2g \
                WORLD_STAGE="$WORLD_STAGE" \
                TARGET=amd64 TARGET_ARCH=amd64 \
                ${QEMUSTATIC:+QEMUSTATIC="$QEMUSTATIC"} \
                2>&1 | tee /tmp/smolbsd-build.log

          - name: Copy artifact to stable path
            run: |
              # Find the built qcow2 (location varies by FreeBSD release Makefile version)
              ARTIFACT=$(find "$FREEBSD_SRC/release" -name "*.qcow2" -newer /tmp/smolbsd-build.log 2>/dev/null | head -1)
              if [ -z "$ARTIFACT" ]; then
                # Fallback: search broader
                ARTIFACT=$(find /home/studio/bsd-build -name "*.qcow2" 2>/dev/null | head -1)
              fi
              if [ -z "$ARTIFACT" ]; then
                echo "ERROR: no qcow2 artifact found after build"
                tail -50 /tmp/smolbsd-build.log
                exit 1
              fi
              echo "Found artifact: $ARTIFACT"
              cp -v "$ARTIFACT" "$OUTPUT_IMAGE"

          - name: Verify image size (must be < 512 MiB)
            run: |
              ACTUAL=$(qemu-img info --output=json "$OUTPUT_IMAGE" | python3 -c \
                "import json,sys; info=json.load(sys.stdin); print(info.get('actual-size', 0))")
              LIMIT=536870912
              echo "actual-size: $ACTUAL bytes ($(echo "$ACTUAL / 1048576" | bc -l | xargs printf '%.1f') MiB)"
              if [ "$ACTUAL" -ge "$LIMIT" ]; then
                echo "FAIL: image size $ACTUAL >= $LIMIT (512 MiB limit)"
                exit 1
              fi
              echo "PASS: image size within 512 MiB budget"

          - name: Write SHA256 manifest
            run: |
              sha256sum "$OUTPUT_IMAGE" > "${OUTPUT_IMAGE}.sha256"
              echo "SHA256 manifest:"
              cat "${OUTPUT_IMAGE}.sha256"
              echo ""
              echo "Build metadata:"
              echo "KERNCONF=SMOLBSD VM_EXTRA_PACKAGES=tpm2-tools VMSIZE=2g" \
                > "$OUTPUT_DIR/BUILD-INFO.txt"
              date >> "$OUTPUT_DIR/BUILD-INFO.txt"
              git rev-parse HEAD >> "$OUTPUT_DIR/BUILD-INFO.txt"
              cat "$OUTPUT_DIR/BUILD-INFO.txt"

          - name: Upload build log on failure
            if: failure()
            run: tail -100 /tmp/smolbsd-build.log || true
    ```

    Write this file completely. Key requirements:

    - timeout-minutes: 120 (build takes 30-90 min on KVM)
    - qemu-img must be available; if not: sudo apt-get install -y qemu-utils
